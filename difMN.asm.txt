.data
msg1: .asciiz "Digite o primeiro valor: "
msg2: .asciiz "Digite o segundo valor: "
msg3: .asciiz "A diferença entre o maior e o menor é: "
.text
.globl main
main:
	li $v0, 4
	la $a0, msg1
	syscall
	
	li $v0, 5
	syscall
	move $t0, $v0
	
	li $v0, 4
	la $a0, msg2
	syscall
	
	li $v0, 5
	syscall
	move $t1, $v0
	
	bgt $t0, $t1, t0_maior
	bgt $t1, $t0, t1_maior
	
	
	
	t0_maior:
		sub $t4, $t0,$t1
		j fim_comparacao
	t1_maior:
		sub $t4, $t1,$t0
	fim_comparacao:
	
	
	li $v0, 4
    	la $a0, msg3
    	syscall
	
	li $v0, 1
	move $a0, $t4     
	syscall
	
	li $v0, 10
